# strategystudio.org
# For now nginx serves pre-rendered pages from ~/np-explorer/public
# (scripts/render_static.py) and assets straight from ~/np-explorer/static.
# When the FastAPI app ships, swap the "location /" block for a proxy_pass
# to the np_explorer upstream and keep the /static and root-file locations.

upstream np_explorer {
    server 127.0.0.1:8000;
}

server {
    server_name strategystudio.org www.strategystudio.org;

    listen [::]:443 ssl ipv6only=on; # managed by Certbot
    listen 443 ssl; # managed by Certbot
    ssl_certificate /etc/letsencrypt/live/strategystudio.org/fullchain.pem; # managed by Certbot
    ssl_certificate_key /etc/letsencrypt/live/strategystudio.org/privkey.pem; # managed by Certbot
    include /etc/letsencrypt/options-ssl-nginx.conf; # managed by Certbot
    ssl_dhparam /etc/letsencrypt/ssl-dhparams.pem; # managed by Certbot

    root /home/michael/np-explorer/public;
    index index.html;
    charset utf-8;
    client_max_body_size 1m;

    # No add_header inside locations below: that would drop these headers.
    add_header X-Content-Type-Options "nosniff" always;
    add_header X-Frame-Options "DENY" always;
    add_header Referrer-Policy "same-origin" always;
    add_header Content-Security-Policy "default-src 'self'; base-uri 'self'; form-action 'self'; frame-ancestors 'none'; object-src 'none'" always;

    error_page 404 /404.html;
    error_page 403 =404 /404.html;
    error_page 500 502 503 504 /500.html;

    location / {
        expires 5m;
        try_files $uri $uri.html $uri/ =404;
    }

    # Until the app exists, a search shows the landing page.
    location = /search {
        expires 5m;
        try_files /index.html =404;
    }

    location = /404.html { internal; }
    location = /500.html { internal; }

    location /static/ {
        alias /home/michael/np-explorer/static/;
        expires 7d;
    }

    location ~ ^/(favicon\.ico|favicon\.svg|apple-touch-icon\.png|robots\.txt|sitemap\.xml)$ {
        root /home/michael/np-explorer/static;
        expires 7d;
    }

    location = /site.webmanifest {
        root /home/michael/np-explorer/static;
        default_type application/manifest+json;
        expires 7d;
    }
}

server {
    if ($host = www.strategystudio.org) {
        return 301 https://$host$request_uri;
    } # managed by Certbot

    if ($host = strategystudio.org) {
        return 301 https://$host$request_uri;
    } # managed by Certbot

    listen 80;
    listen [::]:80;
    server_name strategystudio.org www.strategystudio.org;
    return 404; # managed by Certbot
}
