# strategystudio.org
# The FastAPI app (np-explorer.service, 127.0.0.1:8000) serves every page.
# nginx serves /static, the root icon/robots files and the nightly sitemaps
# itself, and falls back to the pre-rendered public/500.html if the app is down.
# Rate limits, connection limits and the bad-agent list: np-limits.conf, np-bad-agents.conf.

upstream np_explorer {
    server 127.0.0.1:8000;
}

server {
    server_name strategystudio.org www.strategystudio.org;

    listen [::]:443 ssl ipv6only=on; # managed by Certbot
    listen 443 ssl; # managed by Certbot
    ssl_certificate /etc/letsencrypt/live/strategystudio.org/fullchain.pem; # managed by Certbot
    ssl_certificate_key /etc/letsencrypt/live/strategystudio.org/privkey.pem; # managed by Certbot
    include /etc/letsencrypt/options-ssl-nginx.conf; # managed by Certbot
    ssl_dhparam /etc/letsencrypt/ssl-dhparams.pem; # managed by Certbot

    root /home/michael/np-explorer/public;
    index index.html;
    charset utf-8;
    # Small requests only (the contact form is the one POST), short timeouts, 20 connections per IP.
    client_max_body_size 64k;
    client_body_timeout 10s;
    client_header_timeout 10s;
    keepalive_timeout 30s;
    send_timeout 30s;
    limit_conn np_conn 20;

    access_log /var/log/nginx/access.log;
    access_log /var/log/nginx/np-limited.log np_limited if=$np_limited;

    # One canonical host: www redirects to the bare domain (one copy of every page for search engines).
    if ($host = www.strategystudio.org) {
        return 301 https://strategystudio.org$request_uri;
    }

    if ($np_block_ua) {
        return 403;
    }

    # No add_header inside locations below: that would drop these headers.
    add_header X-Content-Type-Options "nosniff" always;
    add_header X-Frame-Options "DENY" always;
    add_header Referrer-Policy "same-origin" always;
    add_header Content-Security-Policy "default-src 'self'; base-uri 'self'; form-action 'self'; frame-ancestors 'none'; object-src 'none'" always;

    error_page 403 =404 /404.html;
    error_page 429 /429.html;
    error_page 502 503 504 /500.html;

    # Search, person and Ask pages (and their JSON): 30 a minute, burst 10. Popups: 120 a minute.
    # Google and Bing crawlers (by published IP range) skip those and get 600 a minute: np-limits.conf.
    location ~ ^/(search|person/|contractors|ask|api/search|api/person/) {
        limit_req zone=np_heavy burst=10 nodelay;
        limit_req zone=np_frag burst=30 nodelay;
        limit_req zone=np_crawl burst=200 nodelay;
        include /etc/nginx/snippets/np-proxy.conf;
    }

    # Every other dynamic page: 120 a minute, burst 30.
    location / {
        limit_req zone=np_dynamic burst=30 nodelay;
        limit_req zone=np_crawl burst=200 nodelay;
        include /etc/nginx/snippets/np-proxy.conf;
    }

    location = /404.html { internal; }
    location = /500.html { internal; }
    location = /429.html {
        internal;
        # add_header here replaces the server-level headers, so repeat them.
        add_header Retry-After 60 always;
        add_header Cache-Control "no-store" always;
        add_header X-Content-Type-Options "nosniff" always;
        add_header X-Frame-Options "DENY" always;
        add_header Referrer-Policy "same-origin" always;
        add_header Content-Security-Policy "default-src 'self'; base-uri 'self'; form-action 'self'; frame-ancestors 'none'; object-src 'none'" always;
    }

    # Sitemap index and per-type sitemaps, rebuilt nightly (loader/sitemap.py).
    location = /sitemap.xml {
        root /data/sitemaps;
        try_files /sitemap.xml @static_sitemap;
        expires 1h;
    }

    location @static_sitemap {
        root /home/michael/np-explorer/static;
        try_files /sitemap.xml =404;
    }

    location /sitemaps/ {
        alias /data/sitemaps/;
        types { application/gzip gz; }
        expires 1h;
    }

    # Static files: no rate limit, long cache. style.css and site.js are linked with ?v=<hash>,
    # so a change reaches browsers at once.
    location /static/ {
        alias /home/michael/np-explorer/static/;
        expires 365d;
    }

    location ~ "^/(favicon\.ico|favicon\.svg|apple-touch-icon\.png|robots\.txt|BingSiteAuth\.xml|[0-9a-f]{32}\.txt)$" {
        root /home/michael/np-explorer/static;
        expires 30d;
    }

    location = /site.webmanifest {
        root /home/michael/np-explorer/static;
        default_type application/manifest+json;
        expires 7d;
    }
}

server {
    if ($host = www.strategystudio.org) {
        return 301 https://strategystudio.org$request_uri;
    } # managed by Certbot

    if ($host = strategystudio.org) {
        return 301 https://$host$request_uri;
    } # managed by Certbot

    listen 80;
    listen [::]:80;
    server_name strategystudio.org www.strategystudio.org;
    return 404; # managed by Certbot
}
