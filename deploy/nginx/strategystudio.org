# strategystudio.org
# The FastAPI app (np-explorer.service, 127.0.0.1:8000) serves every page.
# nginx serves /static, the root icon/robots files and the nightly sitemaps
# itself, and falls back to the pre-rendered public/500.html if the app is down.

upstream np_explorer {
    server 127.0.0.1:8000;
}

server {
    server_name strategystudio.org www.strategystudio.org;

    listen [::]:443 ssl ipv6only=on; # managed by Certbot
    listen 443 ssl; # managed by Certbot
    ssl_certificate /etc/letsencrypt/live/strategystudio.org/fullchain.pem; # managed by Certbot
    ssl_certificate_key /etc/letsencrypt/live/strategystudio.org/privkey.pem; # managed by Certbot
    include /etc/letsencrypt/options-ssl-nginx.conf; # managed by Certbot
    ssl_dhparam /etc/letsencrypt/ssl-dhparams.pem; # managed by Certbot

    root /home/michael/np-explorer/public;
    index index.html;
    charset utf-8;
    client_max_body_size 1m;

    # No add_header inside locations below: that would drop these headers.
    add_header X-Content-Type-Options "nosniff" always;
    add_header X-Frame-Options "DENY" always;
    add_header Referrer-Policy "same-origin" always;
    add_header Content-Security-Policy "default-src 'self'; base-uri 'self'; form-action 'self'; frame-ancestors 'none'; object-src 'none'" always;

    error_page 403 =404 /404.html;
    error_page 502 503 504 /500.html;

    location / {
        proxy_pass http://np_explorer;
        proxy_http_version 1.1;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
        proxy_read_timeout 60s;
        # The app renders its own 404 and 500 pages; nginx only covers it being down.
        proxy_intercept_errors off;
    }

    location = /404.html { internal; }
    location = /500.html { internal; }

    # Sitemap index and per-type sitemaps, rebuilt nightly (loader/sitemap.py).
    location = /sitemap.xml {
        root /data/sitemaps;
        try_files /sitemap.xml @static_sitemap;
        expires 1h;
    }

    location @static_sitemap {
        root /home/michael/np-explorer/static;
        try_files /sitemap.xml =404;
    }

    location /sitemaps/ {
        alias /data/sitemaps/;
        types { application/gzip gz; }
        expires 1h;
    }

    location /static/ {
        alias /home/michael/np-explorer/static/;
        expires 7d;
    }

    location ~ ^/(favicon\.ico|favicon\.svg|apple-touch-icon\.png|robots\.txt)$ {
        root /home/michael/np-explorer/static;
        expires 7d;
    }

    location = /site.webmanifest {
        root /home/michael/np-explorer/static;
        default_type application/manifest+json;
        expires 7d;
    }
}

server {
    if ($host = www.strategystudio.org) {
        return 301 https://$host$request_uri;
    } # managed by Certbot

    if ($host = strategystudio.org) {
        return 301 https://$host$request_uri;
    } # managed by Certbot

    listen 80;
    listen [::]:80;
    server_name strategystudio.org www.strategystudio.org;
    return 404; # managed by Certbot
}
