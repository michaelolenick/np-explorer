# dev.strategystudio.org: the development box (np-explorer-dev).
# Same app, same database as production, behind HTTP basic auth and kept out of
# search engines. Credentials: /etc/nginx/dev.htpasswd (user "dev").
# Copy to /etc/nginx/sites-available/, link into sites-enabled, reload nginx.

upstream np_explorer {
    server 127.0.0.1:8000;
}

server {
    server_name dev.strategystudio.org;

    root /home/michael/np-explorer/public;
    charset utf-8;
    client_max_body_size 64k;

    # No add_header inside locations below: that would drop these headers.
    add_header X-Robots-Tag "noindex, nofollow" always;
    add_header X-Content-Type-Options "nosniff" always;
    add_header X-Frame-Options "DENY" always;
    add_header Referrer-Policy "same-origin" always;
    add_header Content-Security-Policy "default-src 'self'; base-uri 'self'; form-action 'self'; frame-ancestors 'none'; object-src 'none'" always;

    error_page 502 503 504 /500.html;

    # Basic auth goes on each location rather than the server, so Let's Encrypt's
    # challenge location (added by certbot) and robots.txt stay reachable.
    location / {
        auth_basic "strategystudio dev";
        auth_basic_user_file /etc/nginx/dev.htpasswd;
        proxy_pass http://np_explorer;
        proxy_http_version 1.1;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
        proxy_read_timeout 60s;
        proxy_intercept_errors off;
    }

    location /static/ {
        auth_basic "strategystudio dev";
        auth_basic_user_file /etc/nginx/dev.htpasswd;
        alias /home/michael/np-explorer/static/;
    }

    location ~ ^/(favicon\.ico|favicon\.svg|apple-touch-icon\.png)$ {
        root /home/michael/np-explorer/static;
    }

    location = /robots.txt {
        default_type text/plain;
        return 200 "User-agent: *\nDisallow: /\n";
    }

    location = /500.html { internal; }

    listen [::]:443 ssl ipv6only=on; # managed by Certbot
    listen 443 ssl; # managed by Certbot
    ssl_certificate /etc/letsencrypt/live/dev.strategystudio.org/fullchain.pem; # managed by Certbot
    ssl_certificate_key /etc/letsencrypt/live/dev.strategystudio.org/privkey.pem; # managed by Certbot
    include /etc/letsencrypt/options-ssl-nginx.conf; # managed by Certbot
    ssl_dhparam /etc/letsencrypt/ssl-dhparams.pem; # managed by Certbot

}


server {
    if ($host = dev.strategystudio.org) {
        return 301 https://$host$request_uri;
    } # managed by Certbot


    listen 80;
    listen [::]:80;
    server_name dev.strategystudio.org;
    return 404; # managed by Certbot


}
